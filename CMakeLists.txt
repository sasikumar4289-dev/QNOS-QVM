cmake_minimum_required(VERSION 3.16)
project(QNOS_QVM VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(qnos-qvm src/main.cpp)
target_compile_options(qnos-qvm PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_test(NAME bell COMMAND qnos-qvm run ${CMAKE_CURRENT_SOURCE_DIR}/examples/bell.qisa --shots 256 --seed 7 --verify bell)
add_test(NAME ghz5 COMMAND qnos-qvm run ${CMAKE_CURRENT_SOURCE_DIR}/examples/ghz5.qisa --shots 256 --seed 7 --verify ghz)
add_test(NAME hh COMMAND qnos-qvm run ${CMAKE_CURRENT_SOURCE_DIR}/examples/hh.qisa --shots 128 --verify zero)
add_test(NAME dj_constant COMMAND qnos-qvm run ${CMAKE_CURRENT_SOURCE_DIR}/examples/dj_constant.qisa --shots 128 --verify zero)
add_test(NAME dj_balanced COMMAND qnos-qvm run ${CMAKE_CURRENT_SOURCE_DIR}/examples/dj_balanced.qisa --shots 128 --verify one)
add_test(NAME invalid_rejected COMMAND qnos-qvm run ${CMAKE_CURRENT_SOURCE_DIR}/tests/bad_double_alloc.qisa)
set_tests_properties(invalid_rejected PROPERTIES WILL_FAIL TRUE)
